#!/bin/bash

set -euo pipefail

SOURCE="$HOME/backup_lab/source/"
BACKUP_ROOT="$HOME/backup_lab/backups"

DAILY="$BACKUP_ROOT/daily"
MONTHLY="$BACKUP_ROOT/monthly"
YEARLY="$BACKUP_ROOT/yearly"

TODAY=$(date +%Y-%m-%d)
DAY=$(date +%d)
MONTH=$(date +%m)

cleanup()
{
    echo "Backup operation finished."
}

trap cleanup EXIT

# Create backup directories
mkdir -p "$DAILY" "$MONTHLY" "$YEARLY"

echo "Starting backup..."
echo "Source: $SOURCE"

# Daily backup
mkdir -p "$DAILY/$TODAY"

rsync -av --delete \
    "$SOURCE" \
    "$DAILY/$TODAY/"

echo "Daily backup completed: $DAILY/$TODAY"

# Monthly snapshot - first day of month
if [ "$DAY" = "01" ]
then
    mkdir -p "$MONTHLY/$TODAY"

    rsync -av --delete \
        "$SOURCE" \
        "$MONTHLY/$TODAY/"

    echo "Monthly backup created."
fi

# Yearly snapshot - January 1
if [ "$DAY" = "01" ] && [ "$MONTH" = "01" ]
then
    mkdir -p "$YEARLY/$TODAY"

    rsync -av --delete \
        "$SOURCE" \
        "$YEARLY/$TODAY/"

    echo "Yearly backup created."
fi

# Retention policy
find "$DAILY" \
    -mindepth 1 -maxdepth 1 \
    -type d -mtime +7 \
    -exec rm -rf {} \;

find "$MONTHLY" \
    -mindepth 1 -maxdepth 1 \
    -type d -mtime +30 \
    -exec rm -rf {} \;

find "$YEARLY" \
    -mindepth 1 -maxdepth 1 \
    -type d -mtime +365 \
    -exec rm -rf {} \;

echo "Retention cleanup completed."

echo
echo "BACKUP REPORT"
echo "----------------------------"
echo "Date   : $(date)"
echo "Source : $SOURCE"
echo "Backup : $DAILY/$TODAY"
echo "Status : SUCCESS"
echo "----------------------------"
